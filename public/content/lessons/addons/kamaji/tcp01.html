kubectl apply -k ./tcp

查看 tcp#

注： 下文生成worker节点接入集群的命令时需要使用这里的 name: tcp01

root@mc01:~# k get tcp
NAME    VERSION   INSTALLED VERSION   STATUS   CONTROL-PLANE ENDPOINT   KUBECONFIG               DATASTORE           AGE
tcp01   v1.32.5   v1.32.5             Ready    192.168.3.102:6443       tcp01-admin-kubeconfig   tcp01-kamaji-etcd   52s

获取 Cluster A 的 Kubeconfig：#

kubectl get secrets -n tcp01 tcp01-admin-kubeconfig -o json | jq -r '.data["admin.conf"]' | base64 --decode > tcp01.yaml

添加 worker 节点#

  • 生成接入集群的命令
root@mc01:~# kubeadm --kubeconfig=tcp01.yaml token create --print-join-command
kubeadm join 192.168.3.102:6443 --token nef29m.6u18hra5jmnjs5o8 --discovery-token-ca-cert-hash sha256:912c1b3d29b1ae250f1fbbdd9a1693333680b58ddef1d289818d3be288a01a2f

token 默认有效期： 24h

  • 复制安装脚本到所有 worker 节点
cd ./worker
scp yaki.sh containerd.yaml generate_containerd_certs.py root@192.168.3.50:/root
  • 接入集群 （worker 节点执行）
env \
  JOIN_URL="192.168.3.102:6443" \
  JOIN_TOKEN="nef29m.6u18hra5jmnjs5o8" \
  JOIN_TOKEN_CACERT_HASH="sha256:912c1b3d29b1ae250f1fbbdd9a1693333680b58ddef1d289818d3be288a01a2f" \
  KUBERNETES_VERSION="v1.32.5" \
  FILES_REPO="http://file.home.lan:8080/k8s/v1.32.5/" \
  /root/yaki.sh join
  • 配置 containerd (worker节点)
./generate_containerd_certs.py
systemctl daemon-reload
systemctl restart containerd.service
  • 配置 kubelet（worker节点）
echo "serverTLSBootstrap: true" >> /var/lib/kubelet/config.yaml
systemctl restart kubelet
  • 授权 csr
for csr in $(kubectl get csr | grep 'kubernetes.io/kubelet-serving' | grep Pending | awk '{print $1}'); do kubectl certificate approve $csr; done
