MacVLAN#

此配置只适合单网卡同一网段的 RoCE 网络场景使用

配置#

# 配置 RDMA 设备
kubectl apply -f nicclusterpolicy.yaml

# 配置 MacVLAN IP 池
kubectl apply -f hca.yaml

使用与测试#

  • 申请资源必须设置 rdma/hca: 1
  • 设置注解申请 MacVLAN IP k8s.v1.cni.cncf.io/networks: network-operator/hca
# 部署测试 pod
kubectl apply -f tests.yaml

# 进入 server 测试 pod
kubectl exec -it tests-rdma-hca-xxx -- /bin/bash
# 查询 ip 作为 server ip
> ip addr show eth0
# 查询 gid 和 mlx 设备名称
> show_gids |grep v2
# 启动测试, 作为 server, -x 指定 gid，-d 指定 mlx 设备名称
> ib_send_bw -x 7 -d mlx5_bond_0

# 进入 client 测试 pod
kubectl exec -it tests-rdma-hca-yyy -- /bin/bash
# 查询 gid 和 mlx 设备名称
> show_gids |grep v2
# 启动测试, 作为 client，-x 指定 gid，-d 指定 mlx 设备名称
> ib_send_bw -x 7 -d mlx5_bond_0 <server_ip>

# 删除测试 pod
kubectl delete -f tests.yaml

卸载#

kubectl delete -f hca.yaml
kubectl delete -f nicclusterpolicy.yaml
