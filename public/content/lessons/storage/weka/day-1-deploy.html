Weka 部署#

BIOS 设置#

  • 关闭 CPU 省电模式
    • Procesor Configuration
      • Power Management
        • P States
          • Speed Step: Disabled
        • C States
          • C1E: Disabled
          • Enable Monitor MWAIT: Disabled
  • 关闭 IOMMU
    • IIO Configuration
      • Intel VT for Directed I/O: Disabled
  • 关闭超线程
    • Procesor Configuration
      • Hyper-Threading/ Enable LP[Global] : Disabled

部署 WSA#

  • WSA 包含 RockyLinux，MLNX 驱动和 Weka 软件包等，部署方式类似于 RockyLinux 安装
  • 部署时只修改, 不要修改其它选项避免问题：
    • 安装位置
    • 主机名
  • 等待其部署完成自行重启完成
  • OS 启动后使用 nmtui 配置初始网络，以方便 ssh 访问进来

配置系统#

调整分区大小#

缺省安装后，大量空间倍分配给 /home 目录，导致 / 目录空间不足，需要调整分区大小

rsync -avP /home/weka /tmp/ \
    && umount /home \
    && lvremove -y /dev/mapper/wr-home \
    && swapoff -a \
    && lvremove -y /dev/mapper/wr-swap \
    && cp /etc/fstab /etc/fstab.bak \
    && sed -i 's/.*swap.*/#&/' /etc/fstab \
    && sed -i 's/.*home.*/#&/' /etc/fstab \
    && lvextend -l +100%FREE /dev/mapper/wr-root \
    && xfs_growfs /dev/mapper/wr-root \
    && rsync -avP /tmp/weka /home

配置网络#

# 设置 /etc/hosts
cat << EOF >> /etc/hosts
10.254.41.4 sn-10-254-41-4
...
10.254.41.9 sn-10-254-41-10
EOF

# 准备 pdsh hosts file
cat << EOF > srv
sn-10-254-41-[4-9]
EOF
export PDSH_RCMD_TYPE=ssh
export PDSH_SSH_ARGS="-o StrictHostKeyChecking=no"
pdsh -w ^srv hostname
# 更新网卡固件参数
pdsh -w ^srv "mst start"
pdsh -w ^srv 'for MLXDEV in /dev/mst/* ; do mlxconfig -d ${MLXDEV} -y set ADVANCED_PCI_SETTINGS=1 PCI_WR_ORDERING=1; done'
pdsh -w ^srv "mlxconfig -d /dev/mst/mt4123_pciconf0 q |grep -e ADVANCED_PCI_SETTINGS -e PCI_WR_ORDERING"
pdsh -w ^srv "mlxconfig -d /dev/mst/mt4123_pciconf1 q |grep -e ADVANCED_PCI_SETTINGS -e PCI_WR_ORDERING"
pdsh -w ^srv reboot
# 修改 NetworkManger 配置增加 ignore-carrier=*
vim /etc/NetworkManager/NetworkManager.conf
[main]
#plugins=ifcfg-rh
ignore-carrier=*
---
cp /etc/NetworkManager/NetworkManager.conf .
pdcp -w ^srv NetworkManager.conf /etc/NetworkManager
# 配置 arp filter
cat << EOF > 80-arp.conf
net.ipv4.conf.all.arp_filter = 1
net.ipv4.conf.default.arp_filter = 1
net.ipv4.conf.all.arp_announce = 2
net.ipv4.conf.default.arp_announce = 2
net.ipv4.conf.all.arp_ignore = 0
net.ipv4.conf.default.arp_ignore = 0
EOF
pdcp -w ^srv 80-arp.conf /etc/sysctl.d/
pdsh -w ^srv sysctl --system

# 添加 table
cat << EOF >> /etc/iproute2/rt_tables
300        net300
301        net301
EOF
cp /etc/iproute2/rt_tables .
pdcp -w ^srv rt_tables /etc/iproute2/

# 设置 IP & Route & Rule
#!/bin/bash

set -x

readonly host=$(hostname)
readonly seq="${host##*-}"

# CHANGE ME
readonly CONFIGS=$(cat <<- EOF
10.251.12.$seq bond1 ens3 300
10.251.13.$seq bond2 ens6 301
EOF
)
readonly NET="10.251.12.0"
readonly NETMASK=22
readonly GATEWAY=10.251.15.254

while read -r ip bond iface table; do
	nmcli connection add type bond con-name $bond ifname $bond bond.options "mode=802.3ad,xmit_hash_policy=layer3+4"
	nmcli connection add type bond-slave con-name ${iface}f0np0 ifname ${iface}f0np0 master $bond
	nmcli connection add type bond-slave con-name ${iface}f1np1 ifname ${iface}f1np1 master $bond

	nmcli connection modify $bond \
		autoconnect yes \
		ipv4.method manual \
		ipv4.addresses ${ip}/${NETMASK} \
		ipv4.gateway $GATEWAY \
		ipv4.never-default yes \
		ipv6.method disabled \
		ethernet.mtu 9000

	nmcli connection modify $bond \
		ipv4.routes "$NET/${NETMASK} src=$ip table=$table" \
		ipv4.routing-rules "priority $table from $ip table $table"

	nmcli connection up $bond
done <<< "$CONFIGS"
# 更新 NTP
pdsh -w ^srv "sed -i 's/^pool.*/pool ntp.aliyun.com iburst/g' /etc/chrony.conf"
pdsh -w ^srv systemctl enable --now chronyd
pdsh -w ^srv chronyc sources
pdsh -w ^srv timedatectl set-timezone Asia/Shanghai
# check
pdsh -w ^srv yum install -y iperf3
cd /opt/tools/install
# 网络测试
./wekanetperf 10.251.12.{4..9}
./wekanetperf 10.251.13.{4..9}
# checker
./wekachecker 10.251.12.{4..9}
./wekachecker 10.251.13.{4..9}

部署#

# 清空盘
pdsh -w ^srv "echo /dev/nvme{0..23}n1 |xargs -n1 blkdiscard"

# 准备部署
cd /opt/tools/install
# 根据环境生成部署脚本。如果节点不全，需要重新执行一下
# 如果wekaconfig执行失败，出现Unable to parse lscpu output - TPC not found错误。请将LANG设置为en_US.UTF-8
./wekaconfig
# 部署, ssh 无密码 hostname 必须已经接受 host key
pdsh -w sn-10-254-41-[4-9] hostname
./config.sh

# 卸载整个集群
./wekadestroy --yes --force --uninstall sn-10-254-41-{4..9}
# 重新安装 weka 包
pdsh -w ^srv "cd /opt/weka-4.4.6.122/ && ./install.sh"
