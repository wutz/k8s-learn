监控#

# 查看 client 端性能
weka stats realtime -F mode=client
# 查看 backend 端性能
weka stats realtime -F mode=backend
# 登录 deepseek1 组织
weka user login ds1admin --org deepseek1

# 查看组织下所有文件系统的性能数据
weka stats --category fs_stats --param fS:*
# 查询 fs 的 id, 例如 5
weka fs list
# 查询 fs 5 的性能数据
weka stats --category fs_stats --param fS:5
# 查看所有文件系统的吞吐量
weka stats --category fs_stats --stat THROUGHPUT --param fS:5

# 查询所有元数据性能数据
weka stats --category ops --stat OPS
