GPFS快照管理#

参考官方文档： https://www.ibm.com/docs/en/storage-scale/5.2.2?topic=scale-protecting-file-data-using-snapshots

# 在GPFS文件系统目录上创建一个文件
echo "hello world:snap1" > /fs1/fileset1/snapfile1

# 创建第一个快照
mmcrsnapshot fs1 snap1

# 修改文件内容
echo "hello world:snap2" >> /gpfs/fileset1/snapfile1

# 创建第二个快照
mmcrsnapshot fs1 snap2

# 查看当前快照的信息
mmlssnapshot fs1
# 得到如下信息
Snapshots in file system fs1:
Directory                SnapId    Status  Created                   ExpirationTime
snap1                    1         Valid   Thu Mar 12 14:52:08 2026  Thu Mar 12 14:52:08 2026
snap2                    2         Valid   Thu Mar 12 14:52:22 2026  Thu Mar 12 14:52:22 2026

# 把/fs1/fs1set1/snapfile1 删除
# 进入到快照目录查看快照中文件的内容
cd /gpfs/.snapshots
diff -Nur snap1/fs1fset1/snapfile1 snap2/fs1fset1/snapfile1
--- snap1/fs1fset1/snapfile1	2026-03-12 14:51:56.686597974 +0800
+++ snap2/fs1fset1/snapfile1	2026-03-12 14:52:18.374038716 +0800
@@ -1 +1 @@
-hello world: snap1
+hello world: snap2

# 将快照snap1中的文件恢复到文件系统目录
cp /gpfs/.snapshots/snap1/fileset1/snapfile1 /gpfs/fileset1/snapfile1

# 删除快照
mmdelsnapshot fs1 snap1
mmdelsnapshot fs1 snap2
