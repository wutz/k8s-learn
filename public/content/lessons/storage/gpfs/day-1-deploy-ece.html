仲裁或管理器节点规则:

  • 对于单个恢复组，以下仲裁节点规则适用
    • 当 scale-out 节点数为 4 时，quorum 节点数设置为 3
    • 当横向扩展节点数为 5 或 6 时，仲裁节点数设置为 5
    • 当横向扩展节点数为 7 个或更多时，仲裁节点数设置为 7
  • 如果恢复组的数量大于 1 且小于或等于 7，那么 7 个仲裁节点将以循环方式分布在各个恢复组中
  • 如果恢复组的数量超过 7 个，则选择 7 个恢复组作为仲裁持有者
  • 如果集群配置中未定义恢复组或仲裁节点, 那么仲裁节点将根据单个恢复组规则进行分布
  • 如果您要在现有集群中添加新的恢复组，或者想要将新节点添加到现有节点类中，那么安装工具包不会修改现有的仲裁配置

使用安装工具包安装#

从 Fix Central 上的 IBM Storage Scale 页面下载补丁包

cat << 'EOF' >> /etc/profile.d/gpfs.sh
export PATH=/usr/lpp/mmfs/bin:$PATH
EOF
source /etc/profile.d/gpfs.sh

# 安装依赖 
pdsh -w ^all dnf update -y
pdsh -w ^all reboot
dnf -y install ansible 
# GPFS 5.2.3 依赖 Python 版本必须大于 3.8 (EL8 缺省 Python 为 3.6)
pdsh -w ^all dnf install -y python3 python3-dmidecode python3-distro python3-ethtool python38 numactl cpp gcc gcc-c++ elfutils elfutils-devel make kernel-devel kernel-headers rpm-build

# 提取安装包 
./Spectrum_Scale_Erasure_Code-5.x.y.z-x86_64-Linux-install --text-only

cd /usr/lpp/mmfs/5.x.y.z/ansible-toolkit/

# 清除节点和配置
./spectrumscale node clear
./spectrumscale config clear gpfs --all

# 设置类型必须为 ece
./spectrumscale setup -s InstallerNodeIP -st ece

# 设置集群名称
./spectrumscale config gpfs -c gpfs1.example.com

# 在集群定义文件中添加节点
# -so = scale-out node, -q = quorum node 
./spectrumscale node add sn001 -so -m -q 
./spectrumscale node add sn002 -so -m -q 
./spectrumscale node add sn003 -so -m -q 
./spectrumscale node add sn004 -so
# -a = admin node, -g = gui server
# -so 与 -g 不能同时指定
./spectrumscale node add gui01 -a -g 

# 显示集群定义文件中指定的节点列表
./spectrumscale node list

# 执行环境预检查
./spectrumscale callhome disable
./spectrumscale install -pr

# 执行安装工具包安装
./spectrumscale install --skip no-ece-check

# 映射 nvme 插槽位置
pdsh -w ^all 'rm -f /usr/lpp/mmfs/data/gems/*.edf'
mmshutdown -a
# 输入盘对应 slot 编号
# * 每个磁盘符会闪烁亮灯, 然后输入位置编号
# * 查看 nvme 插槽位置: tslsenclslot -ad| mmyfields -s slot SlotHandle LocationCode Devices| grep gems | awk '{print "location: "$2" device: "$3}'
# * 如果是 HDD 则使用 --mode lmr
ecedrivemapping --mode nvme --slotrange 0 23
ecedrivemapping --mode nvme --report
# 拷贝描述文件到其它磁盘布局相同节点
cp /usr/lpp/mmfs/data/gems/*.edf .
pdcp -w ^all *.edf /usr/lpp/mmfs/data/gems/
mmstartup -a

创建文件系统#

# 创建节点类
mmvdisk nc create --nc NC1 -N sn001,sn002,sn003,sn004
mmvdisk nc list --nc all
# 这会触发 daemon 滚动重启
mmvdisk server configure --nc NC1 --recycle one --pagepool 70%
mmvdisk server list --nc NC1 --disk-topology

# 清除GPFS标签和分区表
# 批量清除 nvme0n1 ~ nvme23n1 
for i in {0..23}; do
  echo "Clearing /dev/nvme${i}n1..."
  dd if=/dev/zero of=/dev/nvme${i}n1 bs=1M count=1024 status=progress conv=fsync
done

# 创建恢复组, nvme 插槽映射必须完成
mmvdisk rg create --rg RG1 --nc NC1
mmvdisk rg list --da

# 创建元数据和数据分离的 vdisk set。--set-size参数可以指定百分比（如：90%）也可以指定大小（如：50T）.
mmvdisk vs define --vs fs1meta --rg RG1 --code 4WayReplication --bs 512K --da DA1 --nsd-usage metadataonly --sp system --set-size 5%
mmvdisk vs define --vs fs1vs1 --rg RG1 --code 8+3P --bs 4M --da DA1 --nsd-usage dataonly --sp data --set-size 90%
mmvdisk vs create --vs fs1meta
mmvdisk vs create --vs fs1vs1
# 创建元数据和数据混合的 vdisk set。 --set-size参数可以指定百分比（如：90%）也可以指定大小（如：50T）.
mmvdisk vs define --vs fs2vs1 --rg RG1 --code 8+3P --bs 4M --da DA1 --nsd-usage dataAndMetadata  --sp system --set-size 50T
mmvdisk vs define --vs fs2vs2 --rg RG1 --code 8+3P --bs 4M --da DA1 --nsd-usage dataAndMetadata  --sp system --set-size 50T
mmvdisk vs define --vs fs2vs3 --rg RG1 --code 8+3P --bs 4M --da DA1 --nsd-usage dataAndMetadata  --sp system --set-size 50T
mmvdisk vs create --vs fs2vs1
mmvdisk vs create --vs fs2vs2
mmvdisk vs create --vs fs2vs3
mmvdisk vs list --vs all

# 创建文件系统 fs1
mmvdisk fs create --fs fs1 --vs fs1meta,fs1vs1 --mmcrfs -T /fs1
mmchfs fs1 --auto-inode-limit
mmchfs fs1 --perfileset-quota
mmchfs fs1 --filesetdf
mmchfs fs1 -Q yes

mmmount fs1 -a
# 创建文件系统 fs2
mmvdisk fs create --fs fs2 --vs fs2vs1 --mmcrfs -T /fs2
mmchfs fs2 --auto-inode-limit
mmchfs fs2 --perfileset-quota
mmchfs fs2 --filesetdf
mmchfs fs2 -Q yes
mmmount fs2 -a

文件系统扩缩容#

文件系统扩缩容： https://www.ibm.com/docs/en/storage-scale-ece/5.2.3?topic=command-mmvdisk-filesystem

扩容

# 将剩余的vs fs2vs2 加入到文件系统 fs2 实现扩容
mmvdisk filesystem add --file-system fs2 --vdisk-set fs2vs2

# 平衡文件系统。否则会在 mmhealth cluster show node 中显示 ill_unbalanced_fs 警告
mmrestripefs fs2 -b

缩容

# 将剩余的vs fs2vs2 文件系统 fs2 移除实现缩容
mmvdisk filesystem delete --file-system fs2 --vdisk-set fs2vs2

# 平衡文件系统。否则会在 mmhealth cluster show node 中显示 ill_unbalanced_fs 警告
mmrestripefs fs2 -b
