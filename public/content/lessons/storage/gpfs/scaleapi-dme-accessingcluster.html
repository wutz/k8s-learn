使用 ScaleAPI 部署 DME Accessing Cluster#

  • 版本大于等于 5.2.3.0

概念#

部署#

  • 配置 ssh 无密码登录
    • 配置 ansible node 到 installer node
    • 配置所有 admin node 到所有 node
    • 通常 installer node 是第一个 admin node
  • 以下在执行 ansible 安装节点执行
# DEB
apt install ansible-core
# or RPM
dnf install ansible-core

cat << 'EOF' >> /etc/profile.d/gpfs.sh
export PATH=/usr/lpp/mmfs/bin:$PATH
EOF
source /etc/profile.d/gpfs.sh

# 解压开安装包, 输入 1 接收协议
./Storage_Scale_Data_Management-5.2.3.*-x86_64-Linux-install

cd /usr/lpp/mmfs/5.2.3.*/ansible-toolkit
./spectrumscale setup -s <installerNodeIP>
# 设置集群名称
./spectrumscale config gpfs -c dev1.bz1.paratera.com
# 禁用 callhome
./spectrumscale callhome disable
# 查看配置
./spectrumscale config gpfs --list
# 开启支持 scaleapi
./spectrumscale scaleadmd enable

./spectrumscale node list
# -q -- quorum node, 使用节点数满足情况下最大数值 3，5，7
# -m -- manager node，与 quorum node 一起设置
# -a -- admin node, 推荐 3
./spectrumscale node add client1 -m -q -a
./spectrumscale node add client2 -m -q -a
./spectrumscale node add client3 -m -q -a

mkdir -p ~/tls
cd ~/tls
# 创建节点根证书
openssl ecparam -name prime256v1 -genkey -noout -out ca.key
openssl req -new -x509 -sha256 -key ca.key -out ca.crt -subj "/C=CN/ST=Beijing/L=Beijing/O=Paratera/CN=`hostname`" -days 10000
# 创建节点
openssl ecparam -name prime256v1 -genkey -noout -out server.key
# CN 必须与集群名称一致
openssl req -new -sha256 -key server.key -out server.csr -subj "/C=CN/ST=Beijing/L=Beijing/O=Paratera/CN=dev1.bz1.paratera.com"
openssl x509 -req -in server.csr -CA ca.crt -CAkey ca.key -CAcreateserial -out server.pem -days 10000 -sha256
cd -
./spectrumscale nodeid define --cert /root/tls/server.pem --key /root/tls/server.key --chain /root/tls/ca.crt -N all

./spectrumscale install --precheck
./spectrumscale install

远程集群#

Remote Cluster Support - Storage Scale Native REST API

# 授权
scalectl cluster remote authorize --name scale1.bz1.paratera.com

# 添加远程集群
scalectl cluster remote add --name scale1.bz1.paratera.com --contact-nodes server1,server2,server3

# 添加远程文件系统
scalectl filesystem remote add --name fs0 -T /fs0 -A yes --remote-name fs0 --owning-cluster scale1.bz1.paratera.com

# 删除远程文件系统
scalectl filesystem remote remove fs0
# 删除远程集群
scalectl cluster remote delete scale1.bz1.paratera.com

文件系统管理#

远程与本地文件系统管理方式完全相同

# 挂载远程文件系统
scalectl filesystem mount fs0 -N all

# 查看挂载状态
scalectl fs mountState fs0

# 卸载远程文件系统
scalectl filesystem unmount fs0
