使用 ScaleAPI 部署 DME Owning Cluster#

  • 版本大于等于 5.2.3.0

概念#

部署#

  • 配置 ssh 无密码登录
    • 配置 ansible node 到 installer node
    • 配置所有 admin node 到所有 node
    • 通常 installer node 是第一个 admin node
  • 以下在执行 ansible 安装节点执行
# DEB
apt install ansible-core
# or RPM
dnf install ansible-core

cat << 'EOF' >> /etc/profile.d/gpfs.sh
export PATH=/usr/lpp/mmfs/bin:$PATH
EOF
source /etc/profile.d/gpfs.sh

# 解压开安装包, 输入 1 接收协议
./Storage_Scale_Data_Management-5.2.3.*-x86_64-Linux-install

cd /usr/lpp/mmfs/5.2.3.*/ansible-toolkit
./spectrumscale setup -s <installerNodeIP>
# 设置集群名称
./spectrumscale config gpfs -c scale1.bz1.paratera.com
# 禁用 callhome
./spectrumscale callhome disable
# 查看配置
./spectrumscale config gpfs --list
# 开启支持 scaleapi
./spectrumscale scaleadmd enable

./spectrumscale node list
# -n -- nsd node, 配置硬盘节点均设置
# -m -- manager node，与 nsd node 一起设置
# -q -- quorum node, 使用节点数满足情况下最大数值 3，5，7
# -a -- admin node, 推荐 3
./spectrumscale node add server1 -m -n -q -a
./spectrumscale node add server2 -m -n -q -a
./spectrumscale node add server3 -m -n -q -a

# 如果磁盘被格式化可以执行 blkdiscard /dev/nvme0n1 -f 清空
./spectrumscale nsd add /dev/nvme0n1 /dev/nvme1n1 -p server1
./spectrumscale nsd add /dev/nvme0n1 /dev/nvme1n1 -p server2
./spectrumscale nsd add /dev/nvme0n1 /dev/nvme1n1 -p server3

mkdir -p ~/tls
cd ~/tls
# 创建节点根证书
openssl ecparam -name prime256v1 -genkey -noout -out ca.key
openssl req -new -x509 -sha256 -key ca.key -out ca.crt -subj "/C=CN/ST=Beijing/L=Beijing/O=Paratera/CN=`hostname`" -days 10000
# 创建节点
openssl ecparam -name prime256v1 -genkey -noout -out server.key
# CN 必须与集群名称一致
openssl req -new -sha256 -key server.key -out server.csr -subj "/C=CN/ST=Beijing/L=Beijing/O=Paratera/CN=scale1.bz1.paratera.com"
openssl x509 -req -in server.csr -CA ca.crt -CAkey ca.key -CAcreateserial -out server.pem -days 10000 -sha256
cd -
./spectrumscale nodeid define --cert /root/tls/server.pem --key /root/tls/server.key --chain /root/tls/ca.crt -N all

./spectrumscale install --precheck
./spectrumscale install

文件系统管理#

scalectl fs create --name fs0 --disks nsd1,nsd2,nsd3 -T /fs0 -Q yes -a --auto-inode-limit --perfileset-quota --filesetdf -V 5.2.3.0 -r 2 -m 2
scalectl fs list
scalectl fs get fs0

# 挂载指定文件系统
scalectl fs mount fs0 -N all
# 挂载所有文件系统
scalectl fs mountAll -N all
# 查看指定挂载状态
scalectl fs mountState fs0

scalectl fs unmount fs0
scalectl fs unmountAll -N all

scalectl fs disks list fs0
scalectl fs disks get fs0 nsd1

scalectl fs pools list fs0

文件集管理#

scalectl fileset create fs0 -n fset1 --inode-space new
scalectl fileset link fs0 fset1 -J /fs0/fset1

scalectl fileset list fs0
scalectl fileset get fs0 fset1 --view long

scalectl fileset unlink fs0 fset1
scalectl fileset delete fs0 fset1

quota 管理#

quota 只能在本拥有集群（owning cluster）上设置：

# 设置大小软限制 95G，硬限制 100G；文件软限制 95K，硬限制 100K
mmsetquota fs0:fset1 --block 95G:100G --files 95K:100K
# 查看 quota 信息
mmlsquota -j fset1 fs0

对 root 用户施加 fileset quota 限制，只需在访问集群（accessing cluster）上执行：

mmchconfig enforceFilesetQuotaOnRoot=yes -i

如果在开启 enforceFilesetQuotaOnRoot 之前已经写入过数据，需要在拥有集群上执行 mmcheckquota fs0 重新核算配额用量，否则已有用量统计不准，可能出现无法写入的情况。

QoS 管理#

mmqos class create fs0 --class c1 --fileset fset1
mmqos throttle create fs0 --pool system --class c1 --maxiops 1000 --maxmbs 100
mmqos throttle list fs0
mmqos filesystem enable fs0

远程集群管理#

主要用于把拥有集群分享给访问集群使用：

  • 分离单一 Scale 集群为多个集群，有利于权限分割
  • 实现多租户时必须选项

Remote Cluster Support - Storage Scale Native REST API

# 创建规则文件，允许访问文件系统 fs0 下的文件集 fset1 给访问集群, 其中文件集设置为可选
cat <<EOF > resourcefile.json
{
  "version": 0,
  "resources": {
    "filesystems": [
      {
        "name": "fs0",
        "disposition": "GRANT",
        "access_type": "rw"
      }
    ],
    "filesets": [
      {
        "name": "root",
        "fs_name": "fs0",
        "disposition": "GRANT"
      },
      {
        "name": "fset1",
        "fs_name": "fs0",
        "disposition": "GRANT"
      }
    ]
  }
}
EOF

# 授权访问集群 dev1.bz1.paratera.com 访问文件系统 fs0 下的文件集 fset1
# > 在访问集群上执行 scalectl cluster list 获取 cluster name 和 cluster id
scalectl cluster remote authorize --name dev1.bz1.paratera.com --cluster-id 9389357423519444966 --file resourcefile.json

# 查看远程集群授权情况
scalectl cluster remote list -v full

scalectl cluster remote unauthorize dev1.bz1.paratera.com
