任意管理节点上执行

## 卸载挂载
mmlsmount all -L
mmumount all -a
mmlsmount all -L

## 删除文件系统
mmvdisk fs list 
mmvdisk fs delete --file-system <fs-name>
mmvdisk fs list

## 删除 vdiskset
mmvdisk vs list
# 删除 vdisk
mmvdisk vs delete --vdisk-set all
# 删除 vdiskset
mmvdisk vs undefine --vdisk-set <vs-name>
mmvdisk vs list

## 删除 recovery group
mmvdisk rg list
mmvdisk rg delete --recovery-group <rg-name>
mmvdisk rg list

## 删除 nodeclass
mmvdisk server list
mmvdisk server unconfigure --nc <nc-name>
mmvdisk nc delete --nc <nc-name>
mmvdisk nc list

## 关闭集群
mmgetstate -a
mmshutdown -a
mmgetstate -a
