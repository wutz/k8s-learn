部署 Ceph RBD#

依赖 部署 Ceph 集群

简介#

Ceph RBD 是 Ceph 提供的块存储服务，常用于数据库、虚拟机、容器等场景。

在 Ceph 集群上创建 RBD Pool#

  1. 创建RBD 存储池

    创建副本 Pool

# 创建副本 pool 用于 rbd
ceph osd pool create rbd01 32 32 rep_ssd --bulk
# (可选) 设置 pool 预计大小 (有助于 PG 数量分配到合理值)
ceph osd pool set rbd01 target_size_bytes 200T
# 查看 pool 配置
ceph osd pool get rbd01 all

# 初始化 pool 用于 rbd
rbd pool init rbd01

创建纠删码池

# 创建副本池用于存储元数据
ceph osd pool create rbd01_metadata 32 32 rep_ssd
# 创建纠删码池用于保存数据
ceph osd pool create rbd01_data erasure ec42_ssd --bulk
# 设置overwrite
ceph osd pool set rbd01_data allow_ec_overwrites true
# 初始化元数据池和纠删码池
rbd pool init rbd01_metadata
rbd pool init rbd01_data
为rbd池配置自动清除回收站策略（容器云和云主机都有泄露的情况）

# 假设设置成从2026-08-13T01:00:00 开始每天清理1次指定的rbd池
rbd trash purge schedule add --pool rbd01 1d 2026-08-13T01:00:00
# 查看自动调度策略
rbd trash purge schedule list --pool rbd01
# every 1d starting at 01:00:00
  1. Ceph支持为存储池创建quota
# 为rbd01设置quota为10T
ceph osd pool set-quota rbd01 max_bytes 10995116277760
# 查看rbd01的quota设定
ceph osd pool get-quota rbd01
  1. 生成 Client 访问 Key
ceph auth get-or-create client.rbd01 mon 'profile rbd' osd 'profile rbd pool=rbd01' mgr 'profile rbd pool=rbd01' |tee /etc/ceph/ceph.client.rbd01.keyring
chmod 600 /etc/ceph/ceph.client.rbd01.keyring

在 Linux 客户端使用#

  1. 安装 rbd 工具
apt install -y ceph-common
rbd --version
  1. 获取 conf 和 key 文件存放到 /etc/ceph 目录下
/etc/ceph/ceph.conf
/etc/ceph/ceph.client.rbd01.keyring

  1. 创建 30G image

    副本池

# 副本池创建 30G image1
rbd create --size 30G rbd01/image1
# 纠删码池创建 30G image1
rbd create rbd01_metadata/image1 --size 30G --data-pool rbd01_data

# 查看
rbd ls rbd01
rbd info rbd01/image1
# 扩容
rbd resize --size 50G rbd01/image1
# 缩容
rbd resize --size 20G rbd01/image1 --allow-shrink
# 永久删除
rbd rm rbd01/image1

# 移动到回收站
rbd trash mv rbd01/image1
# 查看回收站
rbd trash ls rbd01
# 恢复
rbd trash restore rbd01/<id>
# 永久删除
rbd trash rm rbd01/<id>
# 清空回收站 (这将永久删除)
rbd trash purge rbd01

纠删码池

 # 纠删码池创建 30G image1
 rbd create rbd01_metadata/image1 --size 30G --data-pool rbd01_data

 # 查看
 rbd ls rbd01_metadata
 rbd info rbd01_metadata/image1
 # 扩容
 rbd resize --size 50G rbd01_metadata/image1
 # 缩容
 rbd resize --size 20G rbd01_metadata/image1 --allow-shrink
 # 永久删除
 rbd rm rbd01_metadata/image1

 # 移动到回收站
 rbd trash mv rbd01_metadata/image1
 # 查看回收站
 rbd trash ls rbd01_metadata
 # 恢复
 rbd trash restore rbd01_metadata/<id>
 # 永久删除
 rbd trash rm rbd01_metadata/<id>
 # 清空回收站 (这将永久删除)
 rbd trash purge rbd01_metadata
  1. 映射块设备到客户端上
# 副本池映射到本地
rbd device map rbd01/image1
# 纠删码池映射到本地
rbd device map rbd01_metadata/image1
# 查看
rbd device ls
# 副本池取消映射
rbd device unmap rbd01/image1
# 纠删码池取消映射
rbd device unmap rbd01_metadata/image1
  1. 格式化为本地文件系统并挂载使用
mkfs.xfs /dev/rbd0
mkdir /mnt/image1
mount /dev/rbd0 /mnt/image1
