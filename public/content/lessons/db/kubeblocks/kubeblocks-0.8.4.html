Kubeblocks#

卸载#

  1. helmwave down
  2. kubectl delete -f hooks/crds.yaml

备份#

  1. 使用 pvc shared-nvme 作为 BackupRepo
kubectl apply -k hooks/backuprepo/prod
  1. 查询 BackupPolicy
kubectl get backuppolicy <cluster-name>-postgresql-backup-policy -o jsonpath='{.spec.backupMethods[0].name}'
pg-basebackup

定时#

按需#

kubectl get secrets <cluster-name>-conn-credential -o jsonpath='{.data.password}' |base64 -d

cat << 'EOF' > backup.yaml
apiVersion: dataprotection.kubeblocks.io/v1alpha1
kind: Backup
metadata:
  name: mybackup
  namespace: default
  annotations:
    dataprotection.kubeblocks.io/connection-password: <password>
spec:
  backupMethod: pg-basebackup
  backupPolicyName: pg-postgresql-backup-policy
EOF
kubectl apply -f backup.yaml
kubectl get backup mybackup -o jsonpath='{.metadata.annotations.dataprotection\.kubeblocks\.io/connection-password}'
